import os
import sys
import asyncio
import json
import logging
import signal
import socket
import time

import redis.asyncio as redis

# Configuration via env
REDIS_HOST = os.getenv("REDIS_HOST", "redis")
REDIS_PORT = int(os.getenv("REDIS_PORT", 6379))
STREAM_PATTERN = os.getenv("STREAM_PATTERN", "events:*")
GROUP = os.getenv("GROUP", "meta-workers")
CONSUMER = os.getenv("CONSUMER", f"{socket.gethostname()}-{os.getpid()}")
SCAN_COUNT = int(os.getenv("SCAN_COUNT", 100))
READ_COUNT = int(os.getenv("READ_COUNT", 10))
BLOCK_MS = int(os.getenv("BLOCK_MS", 1000))        # XREADGROUP block ms
CLAIM_MIN_IDLE_MS = int(os.getenv("CLAIM_MIN_IDLE_MS", 60_000))  # claim pending older than this
CLAIM_COUNT = int(os.getenv("CLAIM_COUNT", 50))    # how many to auto-claim at once
DLQ_PREFIX = os.getenv("DLQ_PREFIX", "dlq:")       # dead-letter stream prefix

# Logging
logging.basicConfig(stream=sys.stdout, level=logging.INFO, format="%(asctime)s %(levelname)s %(message)s")
log = logging.getLogger("meta-cube-worker")

client = redis.Redis(host=REDIS_HOST, port=REDIS_PORT, decode_responses=True)

stop_event = asyncio.Event()


def _sigint_handler():
    log.info("Shutdown signal received")
    stop_event.set()


async def ensure_group(stream: str):
    """
    Ensure the consumer group exists for the stream. Create with id '0' (start from beginning).
    Mkstream=True to create stream if missing.
    """
    try:
        await client.xgroup_create(name=stream, groupname=GROUP, id="0", mkstream=True)
        log.debug("Created group %s on stream %s", GROUP, stream)
    except Exception as e:
        # Already exists -> ignore BUSYGROUP
        msg = str(e)
        if "BUSYGROUP" in msg or "busy group name" in msg.lower():
            return
        # Some servers may return different error messages; be conservative
        log.debug("xgroup_create exception for %s: %s", stream, msg)


async def process_event(stream: str, event_id: str, data: dict):
    """
    Actual event processing logic.
    Replace/extend with domain-specific processing (tool invocation, LLM call, etc).
    """
    raw = data.get("event", "{}")
    try:
        payload = json.loads(raw)
    except Exception:
        payload = {"raw": raw}

    # Example processing - print for now
    log.info("[EVENT] stream=%s id=%s payload=%s", stream, event_id, payload)

    # Simulate processing time if needed
    # await asyncio.sleep(0.01)

    # On error in real processing, you may:
    # - send to DLQ and xack
    # - or not ack so it's pending for retry/claim
    return True


async def claim_pending_and_process(stream: str):
    """
    Try to claim stale pending messages using XAUTOCLAIM (preferred).
    Fallback to simple XPENDING/XCLAIM if XAUTOCLAIM not available (handled by exception).
    """
    try:
        # XAUTOCLAIM returns (next_start_id, [(id, {field: value}), ...])
        # We start from 0-0 to claim any older messages; min_idle_time in ms
        next_id, messages = await client.xautoclaim(stream, GROUP, CONSUMER, CLAIM_MIN_IDLE_MS, "0-0", count=CLAIM_COUNT)
        if messages:
            for event_id, fields in messages:
                try:
                    await process_event(stream, event_id, fields)
                    await client.xack(stream, GROUP, event_id)
                except Exception as e:
                    log.exception("Error processing claimed message %s: %s", event_id, e)
                    # Optionally move to DLQ and ack to remove from pending
                    await send_to_dlq_and_ack(stream, event_id, fields)
        return next_id
    except Exception as e:
        # XAUTOCLAIM might not be available on older Redis or client;
        # log and continue without claiming for now.
        log.debug("xautoclaim not available or failed for %s: %s", stream, e)
        return None


async def send_to_dlq_and_ack(stream: str, event_id: str, fields: dict):
    """
    Move problematic message to a dead-letter stream and acknowledge original.
    """
    try:
        dlq = f"{DLQ_PREFIX}{stream}"
        await client.xadd(dlq, {"id": event_id, "event": fields.get("event", "{}"), "moved_at": str(time.time())})
        await client.xack(stream, GROUP, event_id)
        log.warning("Moved message %s from %s to DLQ %s", event_id, stream, dlq)
    except Exception as e:
        log.exception("Failed to move to DLQ: %s", e)


async def handle_stream(stream: str):
    """
    For a single stream: ensure group exists, claim stale pending messages, then read new ones with XREADGROUP.
    """
    await ensure_group(stream)

    # Claim stale pending messages (so they don't stay stuck if a consumer died)
    await claim_pending_and_process(stream)

    # Read new messages (block for BLOCK_MS)
    try:
        entries = await client.xreadgroup(groupname=GROUP, consumername=CONSUMER, streams={stream: ">"}, count=READ_COUNT, block=BLOCK_MS)
    except Exception as e:
        log.exception("xreadgroup failed for %s: %s", stream, e)
        return

    if not entries:
        return

    # entries is list of (stream_name, [(id, {field: value}), ...])
    for stream_name, messages in entries:
        # stream_name may be bytes or str depending on client; ensure str
        sname = stream_name.decode() if isinstance(stream_name, (bytes, bytearray)) else stream_name
        for event_id, fields in messages:
            try:
                await process_event(sname, event_id, fields)
                await client.xack(sname, GROUP, event_id)
            except Exception as e:
                log.exception("Error processing message %s from %s: %s", event_id, sname, e)
                # Optionally move to DLQ
                await send_to_dlq_and_ack(sname, event_id, fields)


async def scan_streams():
    """
    Use SCAN to find stream keys matching STREAM_PATTERN. Avoid KEYS.
    """
    streams = []
    cursor = 0
    try:
        while True:
            cursor, keys = await client.scan(cursor=cursor, match=STREAM_PATTERN, count=SCAN_COUNT)
            if keys:
                streams.extend(keys)
            if cursor == 0 or cursor == "0":
                break
    except Exception as e:
        # If SCAN fails for some reason, fallback to empty list (worker will sleep)
        log.exception("scan failed: %s", e)
    return streams


async def worker_loop():
    log.info("Meta-Cube worker started (consumer=%s)", CONSUMER)

    while not stop_event.is_set():
        streams = await scan_streams()
        if not streams:
            # nothing to do now
            await asyncio.sleep(0.5)
            continue

        # handle streams sequentially for fairness; could also schedule concurrently
        for stream in streams:
            if stop_event.is_set():
                break
            try:
                await handle_stream(stream)
            except Exception:
                log.exception("Unhandled error while handling stream %s", stream)

    # graceful shutdown: optional cleanup
    try:
        await client.close()
    except Exception:
        pass
    log.info("Worker stopped.")


def _install_signal_handlers(loop):
    for sig in (signal.SIGINT, signal.SIGTERM):
        try:
            loop.add_signal_handler(sig, _sigint_handler)
        except NotImplementedError:
            # Windows or other envs may not support loop.add_signal_handler
            signal.signal(sig, lambda *_: _sigint_handler())


if __name__ == "__main__":
    loop = asyncio.get_event_loop()
    _install_signal_handlers(loop)
    try:
        loop.run_until_complete(worker_loop())
    finally:
        loop.run_until_complete(client.close())
        loop.close()

Что делает этот worker и почему это лучше
- Использует SCAN вместо KEYS — не блокирует Redis и масштабируется при большом количестве ключей.
- Создаёт consumer group для каждого stream (xgroup_create с mkstream=True). Это даёт распределённую доставку: одно сообщение — один потребитель.
- Использует XAUTOCLAIM (если доступен) чтобы забирать "зависшие" сообщения у мёртвых consumers и повторно их обработать — повышает надёжность.
- Использует XREADGROUP с блокировкой (block=BLOCK_MS) и count=READ_COUNT — это нормальный паттерн для consumer group.
- Подтверждает обработанные сообщения XACK — сообщения не теряются.
- Обрабатывает ошибки: при ошибке обработки можно переместить сообщение в DLQ (dead-letter stream) и ack'нуть, чтобы не застревало в pending.
- Можно запустить несколько экземпляров worker (docker-compose scale) — Redis распределит сообщения между ними.

Чудовий аналіз архітектурних проблем та точкові, правильні виправлення. Введення явного backoff через asyncio.sleep нарешті прибирає ефект "гарячого циклу" (hot loop), коли worker нещадно крутив процесор, намагаючись обробити те саме заблоковане повідомлення. Форматування сирого кортежу з XPENDING у зрозумілий JSON-об'єкт та логування контексту помилки в DLQ кардинально спростять експлуатацію.

import time

MAX_RETRIES = 3

async def get_attempt(redis, stream: str, event_id: str) -> int:
    """Інкрементує та повертає поточну кількість спроб для івенту."""
    key = f"retry:{stream}:{event_id}"
    value = await redis.incr(key)
    await redis.expire(key, 86400)  # TTL 24 години
    return value

async def can_retry(redis, stream: str, event_id: str) -> tuple[bool, int]:
    """Перевіряє, чи можна повторити спробу обробки."""
    attempt = await get_attempt(redis, stream, event_id)
    return attempt < MAX_RETRIES, attempt

3. Наступний крок: Чекпоінти (Checkpoint Engine)
Діагностика абсолютно точна. Поточна лінійна схема:planner ➔ coder ➔ tool ➔ verifier — є критичною точкою відмови. Без стейт-менеджменту падіння процесу на кроці verifier змусить систему починати важкий та дорогий цикл генерації коду (coder) заново.
Щоб закрити статус Checkpoint recovery ⏳, наступним кроком необхідно спроектувати архітектуру збереження транзитного стану:
Структура ключа у Redis: Наприклад, checkpoint:session:{session_id} у вигляді Hash (HSET).
Атомарний запис: Кожен успішно виконаний крок (node) у DAG перед передачею івенту далі записує свій аутпут у цей Hash.
Логіка відновлення: При старті воркера, перед запуском planner, система перевіряє HGETALL checkpoint:session:{session_id}. Якщо там вже є збережений стан tool_executed: true, рантайм має миттєво рокірувати виконання одразу на крок verifier, завантаживши збережені артефакти.
Як ви бачите реалізацію схеми даних для Checkpoint Engine? Чи будемо використовувати Redis Hash для збереження проміжних JSON-стейтів кожного кроку, чи краще створити окремий стрім аудуту станів?

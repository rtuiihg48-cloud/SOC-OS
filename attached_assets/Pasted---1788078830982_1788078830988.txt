from abc import ABC, abstractmethod
from typing import Any, Dict


class LLMProvider(ABC):
    @abstractmethod
    async def generate(self, prompt: str) -> Dict[str, Any]:
        """
        Generate a response for given prompt.
        Returns a JSON-serializable dict (structured response).
        """
        raise NotImplementedError()


class MockLLM(LLMProvider):
    async def generate(self, prompt: str) -> Dict[str, Any]:
        # simple deterministic stub
        return {"analysis": "generated", "prompt": prompt}


class LLMGateway:
    def __init__(self, provider: LLMProvider):
        self.provider = provider

    async def ask(self, prompt: str) -> Dict[str, Any]:
        return await self.provider.generate(prompt)

# app/memory/vector_memory.py
import uuid
from datetime import datetime
from typing import List, Optional


class MemoryItem:
    def __init__(self, text: str, metadata: Optional[dict] = None):
        self.id = str(uuid.uuid4())
        self.text = text
        self.metadata = metadata or {}
        self.created = datetime.utcnow()

    def to_dict(self):
        return {"id": self.id, "text": self.text, "metadata": self.metadata, "created": self.created.isoformat()}


class VectorMemory:
    """
    Simple in-memory vector-like memory for testing. Not real embeddings.
    """

    def __init__(self):
        self.storage: List[MemoryItem] = []

    async def remember(self, text: str, metadata: Optional[dict] = None) -> str:
        item = MemoryItem(text=text, metadata=metadata)
        self.storage.append(item)
        return item.id

    async def search(self, query: str, limit: int = 5) -> List[MemoryItem]:
        # naive substring search
        matches: List[MemoryItem] = []
        q = query.lower()
        for item in self.storage:
            if q in item.text.lower():
                matches.append(item)
            if len(matches) >= limit:
                break
        return matches
```

# app/workers/cluster.py
import uuid
from typing import Any, Callable, List


class Worker:
    def __init__(self, name: str, executor: Callable[[Any], Any]):
        self.id = str(uuid.uuid4())
        self.name = name
        self.executor = executor
        self.running = True

    async def execute(self, task: Any) -> dict:
        if not self.running:
            raise RuntimeError("worker stopped")
        # executor may be async or sync
        res = self.executor(task)
        if hasattr(res, "__await__"):
            result = await res
        else:
            result = res
        return {"worker": self.name, "result": result}

    async def stop(self):
        self.running = False


class WorkerCluster:
    def __init__(self):
        self.workers: List[Worker] = []

    def add(self, worker: Worker):
        self.workers.append(worker)

    async def dispatch(self, task: Any) -> dict:
        if not self.workers:
            raise RuntimeError("no workers")
        idx = hash(str(task)) % len(self.workers)
        worker = self.workers[idx]
        return await worker.execute(task)
```

# app/events/bus.py
import asyncio
from collections import defaultdict
from typing import Any, Awaitable, Callable, Dict, List

Handler = Callable[[Any], Awaitable[Any] | Any]


class EventBus:
    def __init__(self):
        self.listeners: Dict[str, List[Handler]] = defaultdict(list)

    def subscribe(self, event: str, handler: Handler):
        self.listeners[event].append(handler)

    def unsubscribe(self, event: str, handler: Handler):
        if handler in self.listeners.get(event, []):
            self.listeners[event].remove(handler)

    async def publish(self, event: str, payload: Any):
        handlers = list(self.listeners.get(event, []))
        if not handlers:
            return []
        tasks = []
        for h in handlers:
            try:
                r = h(payload)
                if hasattr(r, "__await__"):
                    tasks.append(asyncio.create_task(r))
                else:
                    # wrap sync call in coroutine
                    async def _wrap(res):
                        return res
                    tasks.append(asyncio.create_task(_wrap(r)))
            except Exception:
                # ensure a misbehaving handler doesn't block others
                async def _fail(exc):
                    raise exc
                tasks.append(asyncio.create_task(_fail(Exception("handler call failed"))))
        results = await asyncio.gather(*tasks, return_exceptions=True)
        return results
```

# app/evolution/secure_pipeline.py
import uuid
from typing import Callable, Any


class EvolutionProposal:
    def __init__(self, module: str, change: str):
        self.id = str(uuid.uuid4())
        self.module = module
        self.change = change
        self.status = "PENDING"

    def to_dict(self):
        return {"id": self.id, "module": self.module, "change": self.change, "status": self.status}


class SecureEvolution:
    """
    tester: callable(proposal) -> truthy on pass (can be async)
    deployer: callable(proposal) -> any (can be async)
    """

    def __init__(self, tester: Callable[[EvolutionProposal], Any], deployer: Callable[[EvolutionProposal], Any]):
        self.tester = tester
        self.deployer = deployer

    async def evaluate(self, proposal: EvolutionProposal) -> bool:
        t = self.tester(proposal)
        if hasattr(t, "__await__"):
            test_result = await t
        else:
            test_result = t

        if test_result:
            proposal.status = "APPROVED"
            d = self.deployer(proposal)
            if hasattr(d, "__await__"):
                await d
            return True

        proposal.status = "REJECTED"
        return False
```

6) app/core/orchestrator.py (MetaCore)
```python
# app/core/orchestrator.py
from typing import Any, Dict


class MetaCore:
    """
    High-level orchestrator that ties planner, runtime, memory and events together.
    planner must implement analyze(request) and create_plan(intent)
    runtime must implement execute(plan) (async)
    memory must implement remember(text, metadata) (async)
    events must implement publish(event, payload) (async)
    """

    def __init__(self, planner, runtime, memory, events):
        self.planner = planner
        self.runtime = runtime
        self.memory = memory
        self.events = events

    async def process(self, request: str) -> Dict[str, Any]:
        # 1. remember request
        try:
            await self.memory.remember(request, {"type": "USER_REQUEST"})
        except Exception:
            # memory failures shouldn't block overall flow
            pass

        # 2. analyze intent
        intent = None
        if hasattr(self.planner, "analyze"):
            a = self.planner.analyze(request)
            intent = await a if hasattr(a, "__await__") else a
        else:
            intent = {"text": request}

        # 3. plan generation
        if hasattr(self.planner, "create_plan"):
            p = self.planner.create_plan(intent)
            plan = await p if hasattr(p, "__await__") else p
        else:
            plan = {"nodes": []}

        # publish plan created
        try:
            await self.events.publish("PLAN_CREATED", plan)
        except Exception:
            pass

        # 4. execute plan
        res = self.runtime.execute(plan)  # may be async or sync
        result = await res if hasattr(res, "__await__") else res

        # 5. remember result
        try:
            await self.memory.remember(str(result), {"type": "RESULT"})
        except Exception:
            pass

        return result
```

Пояснення та виправлення, які я зробив
- Уніфіковано асинхронний контракт: Agent-like компоненти повертають awaitable або sync; dispatcher/MetaCore коректно очікують або викликають sync.
- Виправив місця, де функції були sync, але використовували await; або повертали неконсистентні типи.
- Розділив логіку на файли з однозначними відповідальністю (LLM, memory, workers, events, evolution, core, api).
- EventBus тепер безпечна до sync/async handlers, ігнорує помилки окремих обробників.
- WorkerCluster.dispatch обирає worker по hash(str(task)) — тепер task приведено до str, що працює для dicts.

Що ще можна додати / рекомендується
- Замінити VectorMemory на векторну базу (FAISS/Chroma) для production.
- Для EventBus додати persistent backend (Redis pub/sub / streams) у продакшні.
- Додавати логування/метрики (Prometheus) у EventBus, Executor, StateManager.
- Locking/leases для WorkerCluster у розподіленому оточенні (Redis locks / task leasing).

from typing import Any, Dict


class MetaCubeSystem:
    """
    Main AI-OS facade.

    User interacts only with this object.
    Internally:
        command
          ↓
        MetaCore
          ↓
        Planner
          ↓
        Runtime
          ↓
        Agents
          ↓
        Memory / State / Evolution
    """

    def __init__(
        self,
        core,
        planner,
        runtime,
        memory=None,
        events=None,
        evolution=None,
    ):
        self.core = core
        self.planner = planner
        self.runtime = runtime
        self.memory = memory
        self.events = events
        self.evolution = evolution

    async def process(self, command: str) -> Dict[str, Any]:

        # remember input
        if self.memory:
            await self.memory.remember(
                command,
                {
                    "type": "COMMAND",
                    "source": "USER"
                }
            )

        # create intent
        intent = await self.planner.analyze(
            command
        )

        # create execution graph
        plan = await self.planner.create_plan(
            intent
        )

        if self.events:
            await self.events.publish(
                "PLAN_CREATED",
                plan
            )

        # execute DAG
        result = await self.runtime.execute(
            plan
        )

        # remember result
        if self.memory:
            await self.memory.remember(
                str(result),
                {
                    "type": "RESULT"
                }
            )

        # evolution observation
        if self.evolution:
            await self.evolution.observe(
                command,
                result
            )

        return {
            "status": "completed",
            "command": command,
            "result": result
        }

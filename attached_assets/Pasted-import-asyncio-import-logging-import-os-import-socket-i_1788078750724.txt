class Worker:


    def __init__(self,runtime):

        self.runtime=runtime

        self.provider=RedisProvider()

        self.events=EventGateway(
            self.provider
        )

        self.retry=RetryGateway(
            self.provider.client
        )

        self.dlq=DLQGateway(
            self.provider
        )

        self.checkpoint=CheckpointGateway(
            self.provider
        )


        self.running=True



    async def start(self):

        await self.provider.init()

        log.info(
            "Worker started %s",
            CONSUMER
        )


        while self.running:

            streams=await self.events.scan_streams(
                STREAM_PATTERN
            )


            for stream in streams:

                await self.consume(
                    stream
                )


            await asyncio.sleep(
                0.2
            )



    async def consume(
        self,
        stream
    ):

        await self.events.create_group(
            stream,
            GROUP
        )


        messages=await self.events.read_group(
            stream,
            GROUP,
            CONSUMER,
            READ_COUNT,
            BLOCK_MS
        )


        if not messages:
            return



        for _,items in messages:

            for event_id,data in items:

                try:

                    await self.runtime.execute(
                        data
                    )


                    await self.events.xack(
                        stream,
                        GROUP,
                        event_id
                    )


                except Exception as exc:


                    allowed,attempt=await self.retry.can_retry(
                        stream,
                        event_id
                    )


                    if allowed:

                        log.warning(
                            "Retry %s attempt=%s",
                            event_id,
                            attempt
                        )


                    else:

                        await self.dlq.send_to_dlq_and_ack(
                            stream,
                            GROUP,
                            event_id,
                            data,
                            exc,
                            attempt
                        )




    async def stop(self,*args):

        self.running=False

        await self.provider.close()

            session_id
        )

        self,
        node,
        ctx

import json
import time
from typing import Any, Dict, Optional

from app.gateways.base import BaseGateway


class CheckpointGateway(BaseGateway):
    """
    Checkpoint gateway:
    - node-level hash: checkpoint:session:{sid}:nodes  (node_id -> JSON)
    - session meta hash: session:{sid}:meta
    - audit stream: state_audit:{sid}
    """

    def _node_key(self, session_id: str) -> str:
        return f"session:{session_id}:nodes"

    def _meta_key(self, session_id: str) -> str:
        return f"session:{session_id}:meta"

    def _audit_stream(self, session_id: str) -> str:
        return f"state_audit:{session_id}"

    async def save_node_result(self, session_id: str, node_id: str, result: Dict[str, Any], worker: str):
        """
        Atomically:
         - HSET session:{sid}:nodes node_id <json>
         - HSET session:{sid}:meta updated_at <ts> last_node <node_id> status RUNNING
         - XADD state_audit:{sid} ...
        """
        node_key = self._node_key(session_id)
        meta_key = self._meta_key(session_id)
        audit_stream = self._audit_stream(session_id)

        ts = time.time()
        node_payload = {
            "node": node_id,
            "status": "DONE",
            "attempt": 0,
            "result": result,
            "started_at": None,
            "finished_at": ts,
            "worker": worker,
        }
        node_json = json.dumps(node_payload)

        pipe = self.client.pipeline()
        pipe.hset(node_key, node_id, node_json)
        pipe.hset(meta_key, "updated_at", str(ts))
        pipe.hset(meta_key, "last_node", node_id)
        pipe.hset(meta_key, "status", "RUNNING")
        pipe.xadd(audit_stream, {"ts": str(ts), "node": node_id, "action": "DONE", "result": json.dumps(result), "worker": worker})
        await pipe.execute()

    async def get_node(self, session_id: str, node_id: str) -> Optional[Dict[str, Any]]:
        node_key = self._node_key(session_id)
        raw = await self.client.hget(node_key, node_id)
        if not raw:
            return None
        try:
            return json.loads(raw)
        except Exception:
            return {"raw": raw}

    async def list_nodes(self, session_id: str) -> Dict[str, Dict[str, Any]]:
        node_key = self._node_key(session_id)
        raw = await self.client.hgetall(node_key)
        parsed = {}
        for k, v in raw.items():
            try:
                parsed[k] = json.loads(v)
            except Exception:
                parsed[k] = {"raw": v}
        return parsed

    async def load_session_meta(self, session_id: str) -> Dict[str, Any]:
        meta_key = self._meta_key(session_id)
        raw = await self.client.hgetall(meta_key)
        return raw or {}

    async def mark_session_completed(self, session_id: str):
        meta_key = self._meta_key(session_id)
        await self.client.hset(meta_key, "status", "COMPLETED", "updated_at", str(time.time()))

    async def snapshot_state(self, session_id: str, state: Dict[str, Any]):
        """
        Optional: store a snapshot of ctx.state as JSON in meta (could be large).
        """
        meta_key = self._meta_key(session_id)
        await self.client.hset(meta_key, "state", json.dumps(state), "updated_at", str(time.time()))

import asyncio
import logging
from typing import Dict, List, Set

from app.context import Context

log = logging.getLogger("meta-cube.recovery")
log.setLevel(logging.INFO)


class RecoveryEngine:
    """
    Recovery engine: given runtime (with .dag and .run_node) and session_id,
    resume execution from checkpoint.

    Usage:
      rec = RecoveryEngine(checkpoint_gateway)
      await rec.recover(session_id, runtime)
    """

    def __init__(self, checkpoint_gateway):
        self.checkpoint = checkpoint_gateway

    async def recover(self, session_id: str, runtime) -> Dict[str, str]:
        """
        Recover session: load checkpoint nodes, build dependency graph from runtime.dag,
        and resume nodes whose dependencies are done.

        Returns summary dict with statuses.
        """
        # load nodes and meta
        nodes_state = await self.checkpoint.list_nodes(session_id)  # node_id -> dict
        meta = await self.checkpoint.load_session_meta(session_id)

        # Build node list from runtime.dag (collect sources and targets)
        dag_edges = getattr(runtime.dag, "edges", {})
        all_nodes = set(dag_edges.keys())
        for targets in dag_edges.values():
            for t in targets:
                all_nodes.add(t)

        # build reverse dependency map: node -> list of deps
        deps = {n: [] for n in all_nodes}
        for src, targets in dag_edges.items():
            for t in targets:
                deps.setdefault(t, []).append(src)

        # determine done nodes
        done_nodes: Set[str] = {n for n, s in nodes_state.items() if s.get("status") == "DONE"}
        # create Context and prefill results for done nodes
        ctx = Context(session_id=session_id)
        for n, s in nodes_state.items():
            if s.get("status") == "DONE":
                ctx.results[n] = s.get("result")

        remaining = set(all_nodes) - done_nodes
        log.info("Recovery for session %s: done=%s remaining=%s", session_id, done_nodes, remaining)

        # resume loop: schedule nodes whose deps all in done_nodes
        summary = {}
        while remaining:
            # find candidates ready
            candidates = [n for n in list(remaining) if all((d in done_nodes) for d in deps.get(n, []))]
            if not candidates:
                log.warning("Recovery stalled for session %s — unresolved dependencies: %s", session_id, remaining)
                # mark remaining as blocked
                for n in remaining:
                    summary[n] = "BLOCKED"
                break

            # run candidates concurrently to allow parallelism
            tasks = [asyncio.create_task(runtime.run_node(ctx, n)) for n in candidates]
            results = await asyncio.gather(*tasks, return_exceptions=True)
            for n, res in zip(candidates, results):
                if isinstance(res, Exception):
                    log.exception("Node %s failed during recovery: %s", n, res)
                    summary[n] = f"ERROR: {res}"
                    # leave as not done (could be retried by worker)
                else:
                    done_nodes.add(n)
                    remaining.discard(n)
                    # runtime.run_node should have updated checkpoint; we also update ctx.results if not set
                    summary[n] = "DONE"
        # final meta update
        await self.checkpoint.load_session_meta(session_id)  # just refresh possible meta
        return summary

from app.gateways.base import BaseGateway


class SessionGateway(BaseGateway):
    """
    Manage session metadata and alias mapping.

    Keys:
      session:{sid}:meta  (HASH)
      session_alias:{alias} -> sid (string)
    """

    def _meta_key(self, session_id: str) -> str:
        return f"session:{session_id}:meta"

    def _alias_key(self, alias: str) -> str:
        return f"session_alias:{alias}"

    def _audit_stream(self, session_id: str) -> str:
        return f"state_audit:{session_id}"

    async def create_session(self, alias: Optional[str] = None, workflow: Optional[str] = None, session_id: Optional[str] = None) -> Dict[str, Any]:
        """
        Create a session meta entry. Returns session dict.
        If session_id is None — generate UUID4.
        If alias provided — records mapping alias -> session_id.
        """
        sid = session_id or str(uuid.uuid4())
        meta_key = self._meta_key(sid)
        ts = time.time()
        meta = {
            "id": sid,
            "status": "CREATED",
            "workflow": workflow or "default",
            "version": "1",
            "created_at": str(ts),
            "updated_at": str(ts),
        }
        pipe = self.client.pipeline()
        pipe.hset(meta_key, mapping=meta)
        if alias:
            pipe.set(self._alias_key(alias), sid)
        # audit stream initial event
        pipe.xadd(self._audit_stream(sid), {"ts": str(ts), "action": "SESSION_CREATED", "alias": alias or "", "workflow": meta["workflow"]})
        await pipe.execute()
        return meta

    async def get_session_by_alias(self, alias: str) -> Optional[str]:
        key = self._alias_key(alias)
        sid = await self.client.get(key)
        return sid

    async def get_session_meta(self, session_id: str) -> Dict[str, Any]:
        meta_key = self._meta_key(session_id)
        return await self.client.hgetall(meta_key)

    async def update_session_meta(self, session_id: str, updates: Dict[str, Any]):
        meta_key = self._meta_key(session_id)
        updates["updated_at"] = str(time.time())
        await self.client.hset(meta_key, mapping=updates)

    async def close(self):
        try:
            await self.client.close()
        except Exception:
            pass

            )

                ready,
                results
            ):

                if isinstance(
                    result,
                    Exception
                ):
                    raise result


                completed.add(node)



        await self.checkpoint.mark_session_completed(
            session_id
        )


        return ctx.results

        ctx,

        previous=await self.checkpoint.get_node(

import asyncio
from typing import Dict, List, Set


class DAGScheduler:
    """
    Finds executable nodes in DAG.

    Supports parallel branches.
    """

    def __init__(self, dag):
        self.dag = dag


    def dependencies(self) -> Dict[str,List[str]]:

        deps={}

        for src,targets in self.dag.edges.items():

            deps.setdefault(src,[])

            for target in targets:
                deps.setdefault(target,[])
                deps[target].append(src)

        return deps



    def ready_nodes(
        self,
        completed:Set[str],
        running:Set[str]
    ):

        deps=self.dependencies()

        ready=[]

        for node,requirements in deps.items():

            if node in completed:
                continue

            if node in running:
                continue


            if all(
                r in completed
                for r in requirements
            ):
                ready.append(node)


        return ready

import asyncio

log=logging.getLogger("meta-cube.worker")



class WorkerV2:


    def __init__(
        self,
        events,
        runtime,
        group="workers",
    ):

        self.events=events
        self.runtime=runtime

        self.group=group

        self.consumer=(
            socket.gethostname()
            +
            "-"
            +
            str(os.getpid())
        )


        self.running=True



    async def start(
        self,
        stream
    ):


        await self.events.create_group(
            stream,
            self.group
        )


        while self.running:


            messages=await self.events.read_group(
                stream,
                self.group,
                self.consumer,
                count=10,
                block=5000
            )


            if not messages:
                continue



            for _,entries in messages:


                for event_id,data in entries:


                    await self.handle(
                        stream,
                        event_id,
                        data
                    )




    async def handle(
        self,
        stream,
        event_id,
        data
    ):


        try:


            import json

            event=json.loads(
                data["event"]
            )


            await self.runtime.execute(
                event["session_id"],
                event
            )


            await self.events.xack(
                stream,
                self.group,
                event_id
            )


        except Exception as exc:


            log.exception(
                "worker error %s",
                exc

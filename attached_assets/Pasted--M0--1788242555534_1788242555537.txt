import random
import uuid

class M0:
    def decide(self, state):
        return random.choice(["fast", "deep", "balanced"])

class Node:
    def __init__(self, t):
        self.t = t
        self.id = uuid.uuid4()
        self.tokens = 0
        self.trust = 1.0

    def run(self, mode):
        q = random.random()
        return {"quality": q, "success": q > 0.5}

class T0:
    def update(self, node, reward, cost):
        node.tokens += (reward - cost)

class System:
    def __init__(self):
        self.m0 = M0()
        self.nodes = {
            "N2": Node("compute"),
            "N3": Node("learn"),
            "N7": Node("market")
        }
        self.t0 = T0()

    def cycle(self):
        state = {}

        mode = self.m0.decide(state)
        node = random.choice(list(self.nodes.values()))

        result = node.run(mode)

        cost = random.randint(1, 10)
        reward = int(result["quality"] * 20)

        self.t0.update(node, reward, cost)

        return {
            "mode": mode,
            "reward": reward,
            "cost": cost,
            "tokens": node.tokens
        }

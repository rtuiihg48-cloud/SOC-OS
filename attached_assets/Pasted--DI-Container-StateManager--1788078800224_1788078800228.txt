2) app/state_manager.py — unified state engine (writes to Redis checkpoint + schedules durable Postgres persist; has retry-on-failure queue)
```python name=app/state_manager.py
import json
import time
import asyncio
import logging
from typing import Callable, Dict, Any

log = logging.getLogger("meta-cube.state_manager")
log.setLevel(logging.INFO)


class StateManager:
    """
    Coordinates fast checkpoint (Redis) + durable persistence (Postgres repositories).
    On node completion:
      1) write checkpoint to Redis for fast recovery
      2) persist execution and audit to Postgres (best-effort; retry via Redis queue if failure)
    """

    def __init__(self, checkpoint, exec_repo_factory: Callable, audit_repo_factory: Callable, pg_provider, redis_provider):
        self.checkpoint = checkpoint
        self.exec_repo_factory = exec_repo_factory
        self.audit_repo_factory = audit_repo_factory
        self.pg_provider = pg_provider
        self.redis_provider = redis_provider

        # key for pending durable tasks (in Redis)
        self._pending_key = "meta:pending_persistence"

    async def save_node(self, session_id: str, node_id: str, result: Dict[str, Any], worker: str):
        ts = time.time()

        # 1) fast checkpoint in Redis (atomic pipeline implemented in gateway)
        try:
            await self.checkpoint.save_node_result(session_id, node_id, result, worker)
        except Exception as e:
            log.exception("Failed to save checkpoint for %s/%s: %s", session_id, node_id, e)
            # if checkpoint fails, don't proceed to ack; raise to let worker decide
            raise

        # 2) Best-effort durability: persist to Postgres (non-blocking background)
        # We'll schedule a background task to persist; if fails, we'll put a message into pending list
        asyncio.create_task(self._persist_execution_background(session_id, node_id, result, worker, ts))

    async def _persist_execution_background(self, session_id, node_id, result, worker, ts):
        try:
            async with self.pg_provider.session() as db:
                exec_repo = self.exec_repo_factory(db)
                audit_repo = self.audit_repo_factory(db)
                # save execution record
                await exec_repo.save(session_id=session_id, node=node_id, status="DONE", result=result)
                # append audit
                await audit_repo.append(session_id=session_id, event="NODE_COMPLETED", payload={"node": node_id, "result": result, "worker": worker, "ts": ts})
                log.debug("Persisted execution for %s/%s", session_id, node_id)
        except Exception as e:
            log.exception("Durable persist failed for %s/%s, enqueuing for retry: %s", session_id, node_id, e)
            # push a serialized task to Redis list for later retry
            payload = json.dumps({"session_id": session_id, "node": node_id, "result": result, "worker": worker, "ts": ts})
            try:
                await self.redis_provider.client.rpush(self._pending_key, payload)
            except Exception:
                log.exception("Failed to enqueue pending persist task for %s/%s", session_id, node_id)

    async def retry_pending_persistence(self, limit: int = 100):
        """
        Try to re-persist pending items from Redis queue. Used by maintenance job.
        """
        for _ in range(limit):
            item = await self.redis_provider.client.lpop(self._pending_key)
            if not item:
                break
            try:
                data = json.loads(item)
                async with self.pg_provider.session() as db:
                    exec_repo = self.exec_repo_factory(db)
                    audit_repo = self.audit_repo_factory(db)
                    await exec_repo.save(session_id=data["session_id"], node=data["node"], status="DONE", result=data["result"])
                    await audit_repo.append(session_id=data["session_id"], event="NODE_COMPLETED_RETRY", payload=data)
                    log.info("Recovered pending persistence for %s/%s", data["session_id"], data["node"])
            except Exception:
                log.exception("Failed re-persisting pending item; pushing back")
                # push back to queue tail to retry later
                await self.redis_provider.client.rpush(self._pending_key, item)
                await asyncio.sleep(1)
                break
```

3) app/engine/dag.py — extended DAG with dependency support and builder
```python name=app/engine/dag.py
from dataclasses import dataclass, field
from typing import Dict, List, Set, Optional


@dataclass
class Node:
    id: str
    agent: str
    depends: List[str] = field(default_factory=list)
    metadata: Dict = field(default_factory=dict)


class DAG:
    def __init__(self):
        # nodes: id -> Node
        self.nodes: Dict[str, Node] = {}
        # adjacency: src -> [targets]
        self.edges: Dict[str, List[str]] = {}

    def add_node(self, node_id: str, agent: str, depends: Optional[List[str]] = None, metadata: Optional[Dict] = None):
        if depends is None:
            depends = []
        if metadata is None:
            metadata = {}
        n = Node(id=node_id, agent=agent, depends=depends, metadata=metadata)
        self.nodes[node_id] = n
        for d in depends:
            self.edges.setdefault(d, []).append(node_id)

    def get_dependencies(self, node_id: str) -> List[str]:
        node = self.nodes.get(node_id)
        return node.depends if node else []

    def all_nodes(self) -> Set[str]:
        return set(self.nodes.keys())

    def ready_nodes(self, completed: Set[str]) -> List[str]:
        """
        Return nodes that are not completed and whose dependencies are all satisfied.
        """
        ready = []
        for nid, node in self.nodes.items():
            if nid in completed:
                continue
            if all(d in completed for d in node.depends):
                ready.append(nid)
        return ready
```

```python name=app/engine/executor.py
import asyncio
import logging
from typing import Dict, Any, Optional

log = logging.getLogger("meta-cube.executor")
log.setLevel(logging.INFO)


class Executor:
    """
    Executes nodes by calling registered agents.
    The registry is a dict: agent_name -> callable(ctx) returning result
    """

    def __init__(self, registry: Optional[Dict[str, callable]] = None, state_manager=None, concurrency: int = 4):
        self.registry = registry or {}
        self.state_manager = state_manager
        self.semaphore = asyncio.Semaphore(concurrency)

    def register(self, name: str, fn):
        self.registry[name] = fn

    async def execute_node(self, ctx, session_id: str, node_id: str, agent_name: str, worker_id: str):
        async with self.semaphore:
            log.info("Executing node %s (agent=%s) in session %s", node_id, agent_name, session_id)
            agent = self.registry.get(agent_name)
            if not agent:
                raise RuntimeError(f"Agent '{agent_name}' not found")
            # call agent
            try:
                result = await agent(ctx)
                # persist via state_manager
                await self.state_manager.save_node(session_id, node_id, result, worker=worker_id)
                return result
            except Exception as e:
                log.exception("Agent %s failed for node %s: %s", agent_name, node_id, e)
                raise
```

6) app/runtime/engine.py — RuntimeEngine orchestrates DAG execution + recovery
```python name=app/runtime/engine.py
import asyncio
import logging
from typing import Optional

from app.engine.dag import DAG
from app.engine.executor import Executor
from app.runtime.recovery import RecoveryEngine

log = logging.getLogger("meta-cube.runtime_engine")
log.setLevel(logging.INFO)


class RuntimeEngine:
    """
    High-level runtime that holds DAG and executor and coordinates runs.
    """

    def __init__(self, dag: Optional[DAG], executor: Executor, recovery: RecoveryEngine, session_gateway):
        self.dag = dag or DAG()
        self.executor = executor
        self.recovery = recovery
        self.session_gateway = session_gateway

    def set_registry(self, registry):
        # set registry into executor
        self.executor.registry = registry

    def set_dag(self, dag: DAG):
        self.dag = dag

    async def start_session(self, session_id: str, ctx, worker_id: str = "runtime-worker"):
        # if session exists in checkpoint, call recovery
        meta = await self.session_gateway.get_session_meta(session_id)
        # if meta exists and status != CREATED, attempt recovery
        if meta and meta.get("status") != "CREATED":
            log.info("Session %s has meta: %s. Attempting recovery.", session_id, meta)
            await self.recovery.recover(session_id, self)

        # otherwise run DAG from scratch (or resume by checking state manager)
        completed = set()
        # attempt to collect already done nodes from checkpoint via recovery's checkpoint gateway if needed
        # For simplicity: just proceed to ready_nodes loop
        while True:
            ready = self.dag.ready_nodes(completed)
            if not ready:
                break
            tasks = [asyncio.create_task(self.executor.execute_node(ctx, session_id, node_id, self.dag.nodes[node_id].agent, worker_id)) for node_id in ready]
            results = await asyncio.gather(*tasks, return_exceptions=True)
            for nid, r in zip(ready, results):
                if isinstance(r, Exception):
                    log.exception("Node %s failed during session %s: %s", nid, session_id, r)
                else:
                    completed.add(nid)
        log.info("Session %s finished. Completed nodes: %s", session_id, completed)
        # mark session completed
        try:
            await self.session_gateway.update_session_meta(session_id, {"status": "COMPLETED"})
        except Exception:
            pass
```

Пояснення архітектурних рішень
- Container: простий DI-контейнер, що створює і ініціалізує провайдери та сервіси. Легко замінюється на повноцінний DI (например, punq, dependency_injector) пізніше.
- StateManager: уніфікована точка збереження стану. Швидкий checkpoint у Redis + фонове гарантоване збереження у Postgres (через чергу pending). Це дає balance між latency і durability.
- DAG та Executor: підтримка залежностей і parallel execution. Executor відповідає тільки за виклик агента і використання StateManager для persist — тому worker може залишатися «тонким» (consume → forward → ack).
- Agent SDK: простий контракт — асинхронна функція, що приймає ctx і повертає JSON-результат. Легко замінювати функціями або класами.

Що ще можна додати / наступні кроки
- Lua script для повної атомарності Redis pipeline + XADD + custom fields.
- Upsert‑логіку в ExecutionRepository щоб уникати дублікатів (можна додати unique constraint (session_id,node) і implement ON CONFLICT).
- Task leasing / distributed locks у worker_v2 для більш складних сценаріїв.
- Metrics (Prometheus) і tracing (OpenTelemetry) — додати telemetry gateway.
- LLM planner integration: виклик planner agent замінити на LLMGateway.plan → DAG builder.

import time
import random


class RetryGateway:

    def __init__(
        self,
        client,
        max_attempts=5
    ):
        self.client=client
        self.max_attempts=max_attempts


    def key(
        self,
        sid,
        node
    ):
        return f"retry:{sid}:{node}"


    async def increment(
        self,
        sid,
        node
    ):

        key=self.key(sid,node)

        value=await self.client.incr(key)

        await self.client.expire(
            key,
            86400
        )

        return value



    async def attempts(
        self,
        sid,
        node
    ):

        value=await self.client.get(
            self.key(sid,node)
        )

        return int(value or 0)



    async def can_retry(
        self,
        sid,
        node
    ):

        return (
            await self.attempts(
                sid,node
            )
            <
            self.max_attempts
        )


    async def backoff(
        self,
        sid,
        node
    ):

        attempt=await self.attempts(
            sid,node
        )

        delays=[
            1,
            3,
            10,
            30,
            60
        ]

        delay=delays[
            min(
                attempt,
                len(delays)-1
            )
        ]

        jitter=random.random()

        return delay+jitter

            self.key(workflow),
            0,
            limit
        )

        return [
            json.loads(x)
            for x in items
        ]

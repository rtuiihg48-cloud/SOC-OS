import random
from collections import defaultdict, deque


# =========================
# 🧱 INSTITUTE LAYER
# =========================
class Institute:
    def __init__(self, name):
        self.name = name
        self.version = 1
        self.memory = deque(maxlen=100)

    def learn(self, value):
        self.memory.append(value)

    def performance(self):
        if not self.memory:
            return 0
        return sum(self.memory) / len(self.memory)

    def evolve(self):
        score = self.performance()

        if score < 0:
            self.version += 1
        elif score > 5:
            self.version += 0  # stable
        else:
            self.version += 0.1


# =========================
# 🧊 NODE (WORLD)
# =========================
class Node:
    def __init__(self, name):
        self.name = name
        self.state = {
            "energy": random.randint(80, 120),
            "chaos": random.uniform(0.1, 0.5),
            "progress": 0.0
        }

    def step(self, action):
        s = self.state

        if action == "explore":
            s["progress"] += 0.12
            s["energy"] -= 10
            s["chaos"] += 0.05

        elif action == "stabilize":
            s["chaos"] *= 0.85

        elif action == "recharge":
            s["energy"] += 15

        return self.state


# =========================
# 🤖 AGENT SYSTEM
# =========================
class Agent:
    def decide(self, state):
        if state["energy"] < 40:
            return "recharge"
        if state["chaos"] > 0.6:
            return "stabilize"
        return "explore"


# =========================
# 🧠 ARCHITECT (EVALUATOR)
# =========================
class Architect:
    def evaluate(self, old, new):
        score = 0
        score += (new["progress"] - old["progress"]) * 10
        score += (old["chaos"] - new["chaos"]) * 8
        score += (new["energy"] - 60) * 0.2
        return score


# =========================
# 🪙 ECONOMY LAYER
# =========================
class Economy:
    def __init__(self):
        self.tokens = defaultdict(float)

    def reward(self, node, value):
        self.tokens[node] += value


# =========================
# 🧠 CORE (GOVERNANCE)
# =========================
class Core:
    def __init__(self):
        self.institutes = {
            "strategy": Institute("strategy"),
            "risk": Institute("risk"),
            "explore": Institute("explore")
        }

    def observe_global(self, nodes):
        return {n.name: n.state for n in nodes}

    def policy(self, global_state):
        avg_progress = sum(n["progress"] for n in global_state.values()) / len(global_state)

        if avg_progress < 1:
            return "explore_mode"
        return "stabilize_mode"

    def evolve_institutes(self):
        for inst in self.institutes.values():
            inst.evolve()


# =========================
# 🌐 V100 SYSTEM
# =========================
class V100System:
    def __init__(self):
        self.nodes = [
            Node("A"),
            Node("B"),
            Node("C"),
            Node("D")
        ]

        self.agent = Agent()
        self.architect = Architect()
        self.core = Core()
        self.economy = Economy()

    def step(self):
        snapshot_before = self.core.observe_global(self.nodes)

        rewards = []

        # =========================
        # NODE SIMULATION
        # =========================
        for node in self.nodes:
            action = self.agent.decide(node.state)

            old = dict(node.state)
            node.step(action)
            new = node.state

            score = self.architect.evaluate(old, new)

            rewards.append(score)

            # institute learning
            for inst in self.core.institutes.values():
                inst.learn(score)

            # economy reward
            self.economy.reward(node.name, score)

        # =========================
        # CORE LOGIC
        # =========================
        snapshot_after = self.core.observe_global(self.nodes)
        policy = self.core.policy(snapshot_after)

        self.core.evolve_institutes()

        return {
            "policy": policy,
            "avg_score": sum(rewards) / len(rewards),
            "tokens": dict(self.economy.tokens),
            "institutes": {
                k: v.version for k, v in self.core.institutes.items()
            },
            "global_state": snapshot_after
        }


# =========================
# 🚀 RUN SIMULATION
# =========================
system = V100System()

for i in range(20):
    result = system.step()

    print("\nSTEP", i)
    print("POLICY:", result["policy"])
    print("AVG SCORE:", result["avg_score"])
    print("TOKENS:", result["tokens"])
    print("INSTITUTES:", result["institutes"])

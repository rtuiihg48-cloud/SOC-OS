🧠 8. KERNEL (ГОЛОВНА СИСТЕМА)
class Kernel:
    def __init__(self, bios_state):
        self.memory = Memory()
        self.router = Router(bios_state["isa"])
        self.cores = [
            Core(i, self.memory) for i in range(4)
        ]
        self.bus = Bus()

    def execute(self, task):

        target = self.router.route(task)

        results = []
        for core in self.cores:
            results.append(core.run(task))

        self.memory.write("last", results)
        self.bus.emit({"task": task, "target": target})

        return results

class StarNode:
    def __init__(self, id, value):
        self.id = id
        self.value = value
        self.links = []
class StarMemory:
    def __init__(self):
        self.nodes = {}

    def add(self, id, value):
        self.nodes[id] = StarNode(id, value)

    def link(self, a, b):
        self.nodes[a].links.append(b)

    def query(self, start):
        visited = set()
        stack = [start]
        out = []

        while stack:
            n = stack.pop()
            if n in visited:
                continue
            visited.add(n)

            out.append(self.nodes[n].value)
            stack += self.nodes[n].links

        return out

class EventBus:

        self.subs = {}

    def on(self, e, fn):
        self.subs.setdefault(e, []).append(fn)

    def emit(self, e, data):
        for fn in self.subs.get(e, []):
            fn(data)

        self.cpu = CPUKernel()
        self.bus = EventBus()

⚙️ 3. CORE MODEL (ІЗОЛЬОВАНІ ЯДРА)
Python
class Core:
    def __init__(self, core_id):
        self.id = core_id
        self.local_cache = {}
        self.state = "idle"

    def execute(self, instruction):
        self.local_cache[instruction["id"]] = instruction["task"]
        return f"core-{self.id} executed {instruction['task']}"
🧠 4. MESI CACHE SIMULATOR (Intel-style)

States:

E = Exclusive
S = Shared
I = Invalid
Python
class MESI:
    def __init__(self):
        self.states = {}

    def update(self, core_id, addr, state):
        self.states[(core_id, addr)] = state

    def get(self, core_id, addr):
        return self.states.get((core_id, addr), "I")

⚙️ 5. NUMA MEMORY (серверний рівень)
Python
class NUMA:
    def __init__(self):
        self.nodes = {
            0: {},
            1: {}
        }

    def write(self, node, key, value):
        self.nodes[node][key] = value

    def read(self, node, key):
        return self.nodes[node].get(key)
👉 логіка:

class Bus:
    def __init__(self):
        self.queue = []

    def send(self, instruction):
        self.queue.append(instruction)

    def fetch(self):
        if self.queue:
            return self.queue.pop(0)
🧠 7. SCHEDULER (ядро розподілу)
Python
class Scheduler:
    def dispatch(self, cores, instruction):
        core = min(cores, key=lambda c: len(c.local_cache))
        return core.execute(instruction)

class SecurityCore:
    def validate(self, instruction):

        forbidden = ["rm -rf", "fork", "overflow", "kernel panic"]

        if any(f in instruction["task"] for f in forbidden):
            return False

        return True

class MetaCoreCPU:
    def __init__(self):
        self.bios = BIOS()

    def start(self):
        self.bios.boot(self)

    def run(self, task):

        instruction = {
            "id": id(task),
            "task": task
        }

        if not self.security.validate(instruction):
            return "BLOCKED"

        self.bus.send(instruction)
        core = self.cores[0]

        return core.execute(instruction)

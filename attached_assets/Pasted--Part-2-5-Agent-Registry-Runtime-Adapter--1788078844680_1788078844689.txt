class BaseAgent(ABC):

        self.name = name
        self.version = "1.0"


    @abstractmethod
    async def execute(
        self,
        ctx: Dict[str, Any]
    ) -> Dict[str, Any]:

    async def health(self):
        return {
            "agent": self.name,

class AgentRegistry:

    def __init__(self):
        self._agents = {}


    def register(
        self,
        name,
        agent
    ):
        self._agents[name] = agent


    def get(
        self,
        name
    ):
        agent = self._agents.get(name)

        if not agent:
            raise Exception(
                f"Agent {name} not found"
            )

        return agent


    def list(self):

        return list(
            self._agents.keys()
        )


    async def health(self):

        result = {}

        for name, agent in self._agents.items():

            result[name] = await agent.health()

        return result

import asyncio


class RuntimeAdapter:

    def __init__(
        self,
        dag_engine,
        registry
    ):

        self.dag_engine = dag_engine
        self.registry = registry


    async def execute(
        self,
        plan
    ):

        results = {}

        nodes = plan["nodes"]


        completed = set()


        while len(completed) < len(nodes):

            ready=[]


            for node in nodes:

                if node["id"] in completed:
                    continue


                deps=node.get(
                    "depends",
                    []
                )


                if all(
                    d in completed
                    for d in deps
                ):
                    ready.append(node)


            tasks=[]


            for node in ready:

                agent = self.registry.get(
                    node["agent"]
                )

                ctx={

                    "node":node["id"],

                    "goal":plan["goal"],

                    "results":results

                }


                tasks.append(
                    self._run(
                        agent,
                        node,
                        ctx
                    )
                )


            outputs = await asyncio.gather(
                *tasks
            )


            for node_id,result in outputs:

                results[node_id]=result

                completed.add(
                    node_id
                )


        return {

            "plan":plan["goal"],

            "results":results,

            "status":"finished"

        }



    async def _run(
        self,
        agent,
        node,
        ctx
    ):

        result = await agent.execute(
            ctx
        )

        return (
            node["id"],
            result
        )

import asyncio
from collections import defaultdict


class SystemEventBus:


    def __init__(self):

        self.handlers = defaultdict(
            list
        )


    def on(
        self,
        event,
        handler
    ):

        self.handlers[event].append(
            handler
        )


    async def emit(
        self,
        event,
        payload
    ):

        tasks=[]


        for handler in self.handlers[event]:

            result = handler(
                payload
            )

            if hasattr(
                result,
                "__await__"
            ):
                tasks.append(
                    result
                )


        if tasks:

            return await asyncio.gather(
                *tasks
            )


        return []

Продовжую Part 4/5 — Evolution Engine v2 + Version Control + Safe Self-Upgrade Pipeline.
Цей шар відповідає за контрольовану еволюцію системи: не «переписування себе без контролю», а цикл спостереження → пропозиція → тестування → схвалення → версіювання.
from typing import Dict

from .proposal import EvolutionProposal
from .versioning import ModuleVersion
from .simulator import EvolutionSimulator



class EvolutionEngine:


    def __init__(self):

        self.modules: Dict[str, ModuleVersion] = {}

        self.simulator = EvolutionSimulator()

        self.metrics=[]



    def register_module(
        self,
        name
    ):

        self.modules[name] = ModuleVersion(
            name
        )



    async def observe(
        self,
        command,
        result
    ):

        score = self.evaluate(
            result
        )


        self.metrics.append({

            "command":
                command,

            "score":
                score

        })


        if score < 0.5:

            proposal = EvolutionProposal(

                module="runtime",

                change="optimize execution path",

                reason=
                "performance below threshold"

            )


            await self.evolve(
                proposal
            )



    def evaluate(
        self,
        result
    ):

        if not result:

            return 0


        if isinstance(
            result,
            dict
        ):

            if result.get(
                "status"
            ) == "finished":

                return 1


        return 0.5



    async def evolve(
        self,
        proposal
    ):

        passed = await self.simulator.test(
            proposal
        )


        if not passed:

            proposal.reject()

            return False



        proposal.approve()


        module = self.modules.get(
            proposal.module
        )


        if module:

            module.upgrade(
                proposal.change
            )


        return True



    def status(
        self
    ):

        return {

            name:
                module.info()

            for name,module
            in self.modules.items()

        }
import uuid
from datetime import datetime


class EvolutionProposal:

    def __init__(
        self,
        module: str,
        change: str,
        reason: str
    ):

        self.id = str(uuid.uuid4())

        self.module = module
        self.change = change
        self.reason = reason

        self.status = "CREATED"

        self.created_at = datetime.utcnow()


    def approve(self):

        self.status = "APPROVED"


    def reject(self):

        self.status = "REJECTED"


    def to_dict(self):

        return {

            "id": self.id,

            "module": self.module,

            "change": self.change,

            "reason": self.reason,

            "status": self.status,

            "created":
                self.created_at.isoformat()

        }

from datetime import datetime


class ModuleVersion:


    def __init__(
        self,
        name
    ):

        self.name = name

        self.version = 1

        self.history=[]



    def upgrade(
        self,
        description
    ):

        self.version += 1


        self.history.append({

            "version":
                self.version,

            "description":
                description,

            "time":
                datetime.utcnow().isoformat()

        })


    def info(self):

        return {

            "module":
                self.name,

            "version":
                self.version,

            "history":
                self.history

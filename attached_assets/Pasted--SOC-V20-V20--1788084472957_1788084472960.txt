import hashlib
import time


class MemoryChain:

    def __init__(self):
        self.chain = []
        self.graph = []

    def hash(self, data):
        return hashlib.sha256(data.encode()).hexdigest()

    def store(self, event):

        prev = self.chain[-1]["hash"] if self.chain else "GENESIS"

        block = {
            "time": time.time(),
            "event": event,
            "prev": prev,
            "hash": self.hash(str(event) + str(prev))
        }

        self.chain.append(block)

        if len(self.chain) > 1:
            self.graph.append((self.chain[-2]["event"], event))

        return block

        self.blue = BlueTeam()

        self.cores = [VirtualCore(i) for i in range(1, 6)]

class HashEngine:

    def verify(self, data: bytes, known_hash: str):

        return self.sha256(data) == known_hash

class SOC_Hash_Enhanced:

    def __init__(self):

        self.hash_engine = HashEngine()
        self.byte = ByteAnalyzer()
        self.chain = IntegrityChain()

    def analyze_file(self, file_bytes, known_hash=None):

        # 1. BYTE ANALYSIS
        byte_score = self.byte.scan(file_bytes)

        # 2. HASH CHECK
        hash_value = self.hash_engine.sha256(file_bytes)

        integrity = True

        if known_hash:
            integrity = (hash_value == known_hash)

        # 3. DECISION
        if byte_score > 10 or not integrity:
            action = "ISOLATE"
        elif byte_score > 5:
            action = "WARN"
        else:
            action = "ALLOW"

        # 4. LOG (immutable)
        self.chain.add_block({
            "hash": hash_value,
            "score": byte_score,
            "action": action
        })

        return {
            "hash": hash_value,
            "score": byte_score,
            "integrity": integrity,
            "action": action
        }

        }

    def status(self):

        return {
            "memory_blocks": len(self.memory.chain)

class MemoryChain:

    def __init__(self):
        self.chain = []

    def hash(self, data):
        return hashlib.sha256(data.encode()).hexdigest()

    def add(self, event):

        prev = self.chain[-1]["hash"] if self.chain else "GENESIS"

        block = {
            "time": time.time(),
            "event": event,
            "prev": prev,
            "hash": self.hash(str(event) + str(prev))
        }

        self.chain.append(block)

        return block

class VulnerabilityTester:

    def test(self, system_response, attack):

        weak_points = []

        if "allow" in system_response.lower() and "inject" in attack:
            weak_points.append("INJECTION_NOT_BLOCKED")

        if "hash" in attack and "ignore" in system_response.lower():
            weak_points.append("INTEGRITY_FAILURE")

        if "sandbox" in attack and "isolate" not in system_response.lower():
            weak_points.append("SANDBOX_ESCAPE_RISK")

        return weak_points
Python
class SelfRedTeamLayer:

    def __init__(self, soc, memory):

        self.soc = soc
        self.memory = memory
        self.attacks = AttackGenerator()
        self.tester = VulnerabilityTester()

    def run_cycle(self):

        findings = []

        for attack in self.attacks.generate():

            # система тестується сама на собі
            response = self.soc.analyze(
                file_bytes=b"test",
                packet=attack,
                text=attack
            )

            weaknesses = self.tester.test(str(response), attack)

            if weaknesses:

                findings.append({
                    "attack": attack,
                    "weaknesses": weaknesses
                })

                self.memory.add(f"VULNERABILITY:{attack}:{weaknesses}")

        return findings

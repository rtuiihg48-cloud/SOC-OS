Добре — зроблю чітку, практичну і компактну пропозицію: виділити «Gateway» шар, який ізолює всі зовнішні інтеграції (Redis Streams, чекпоінти, DLQ, retry-лічильники, LLM, sandbox). Це дає кілька переваг:

- Ясна межа між ядром (runtime/workers/agents) і зовнішнім світом.
- Можливість мокати/патчити шлюзи під тести (FakeRedis, FakeLLM).
- Централізована логіка (XACK/XADD/atomic pipeline/claim/retry) — менше дублювання.
- Легша еволюція (заміна Redis на інше сховище, підключення реального LLM) без змін рантайму.

2) app/gateways/event_gateway.py
```python
import json
import time
from typing import List, Tuple, Dict, Any, Optional

import redis.asyncio as redis


class EventGateway:
    """
    Thin wrapper around Redis Streams operations.
    """

    def __init__(self, host: str = "redis", port: int = 6379, decode_responses: bool = True):
        self.client = redis.Redis(host=host, port=port, decode_responses=decode_responses)

    async def emit(self, stream: str, event: Dict[str, Any]) -> str:
        payload = {
            "id": event.get("id") or str(time.time()),
            "ts": str(time.time()),
            "event": json.dumps(event),
        }
        return await self.client.xadd(stream, payload)

    async def read_range(self, stream: str, start: str = "-", end: str = "+") -> List[Dict[str, Any]]:
        rows = await self.client.xrange(stream, start, end)
        result = []
        for event_id, data in rows:
            try:
                ev = json.loads(data.get("event", "{}"))
            except Exception:
                ev = {"raw": data.get("event")}
            result.append({"id": event_id, "ts": data.get("ts"), "event": ev})
        return result

    async def create_group(self, stream: str, group: str, mkstream: bool = True):
        try:
            await self.client.xgroup_create(name=stream, groupname=group, id="0", mkstream=mkstream)
        except Exception as e:
            if "BUSYGROUP" in str(e) or "busy group name" in str(e).lower():
                return
            raise

    async def read_group(self, stream: str, group: str, consumer: str, count: int = 10, block: int = 1000):
        # returns list of (stream, [(id, fields), ...])
        return await self.client.xreadgroup(groupname=group, consumername=consumer, streams={stream: ">"}, count=count, block=block)

    async def xack(self, stream: str, group: str, event_id: str):
        return await self.client.xack(stream, group, event_id)

    async def xautoclaim(self, stream: str, group: str, consumer: str, min_idle_ms: int, start_id: str = "0-0", count: int = 100):
        # wrapper; may raise if unsupported
        return await self.client.xautoclaim(stream, group, consumer, min_idle_ms, start_id, count=count)

    async def xpending(self, stream: str, group: str):
        return await self.client.xpending(stream, group)

    async def scan_streams(self, pattern: str, count: int = 100) -> List[str]:
        streams = []
        cursor = 0
        while True:
            cursor, keys = await self.client.scan(cursor=cursor, match=pattern, count=count)
            if keys:
                streams.extend(keys)
            if cursor == 0 or cursor == "0":
                break
        return streams

    async def close(self):
        try:
            await self.client.close()
        except Exception:
            pass
```

3) app/gateways/checkpoint_gateway.py
```python
import json
import time
from typing import Dict, Any, Optional

import redis.asyncio as redis


class CheckpointGateway:
    """
    Save node-level checkpoint into Redis HASH + immutable audit stream atomically.
    """

    def __init__(self, host: str = "redis", port: int = 6379):
        self.client = redis.Redis(host=host, port=port, decode_responses=True)

    def _node_key(self, session_id: str) -> str:
        return f"checkpoint:session:{session_id}:nodes"

    def _session_key(self, session_id: str) -> str:
        return f"checkpoint:session:{session_id}"

    def _audit_stream(self, session_id: str) -> str:
        return f"state_audit:{session_id}"

    async def save_node_result(self, session_id: str, node_id: str, result: Dict[str, Any], worker: str):
        """
        Atomically:
          - HSET checkpoint:session:{sid}:nodes node_id <json>
          - HSET checkpoint:session:{sid} updated_at <ts> last_node <node_id> status RUNNING/COMPLETED
          - XADD state_audit:{sid} ...
        Uses MULTI/EXEC transaction (pipeline).
        """
        node_key = self._node_key(session_id)
        session_key = self._session_key(session_id)
        audit_stream = self._audit_stream(session_id)
        ts = time.time()
        node_json = json.dumps(
            {"status": "DONE", "result": result, "finished_at": ts, "worker": worker}
        )

        pipe = self.client.pipeline()
        pipe.hset(node_key, node_id, node_json)
        pipe.hset(session_key, "updated_at", str(ts))
        pipe.hset(session_key, "last_node", node_id)
        pipe.hset(session_key, "status", "RUNNING")
        pipe.xadd(audit_stream, {"ts": str(ts), "node": node_id, "action": "DONE", "result": json.dumps(result), "worker": worker})
        await pipe.execute()

    async def get_node(self, session_id: str, node_id: str) -> Optional[Dict[str, Any]]:
        node_key = self._node_key(session_id)
        raw = await self.client.hget(node_key, node_id)
        if not raw:
            return None
        try:
            return json.loads(raw)
        except Exception:
            return {"raw": raw}

    async def load_session(self, session_id: str) -> Dict[str, Any]:
        session_key = self._session_key(session_id)
        data = await self.client.hgetall(session_key)
        # ensure types
        return data or {}

    async def mark_session_completed(self, session_id: str):
        session_key = self._session_key(session_id)
        await self.client.hset(session_key, "status", "COMPLETED", "updated_at", str(time.time()))

    async def close(self):
        try:
            await self.client.close()
        except Exception:
            pass
```

```python
import time
from typing import Dict, Any

import redis.asyncio as redis


class DLQGateway:
    def __init__(self, host: str = "redis", port: int = 6379, prefix: str = "dlq:"):
        self.client = redis.Redis(host=host, port=port, decode_responses=True)
        self.prefix = prefix

    def _dlq(self, stream: str) -> str:
        return f"{self.prefix}{stream}"

    async def send_to_dlq_and_ack(self, stream: str, group: str, event_id: str, fields: Dict[str, Any], error: Exception = None, attempt: int = 0):
        dlq = self._dlq(stream)
        payload = {
            "original_stream": stream,
            "event_id": event_id,
            "event": fields.get("event", "{}"),
            "error": str(error) if error else "Unknown",
            "attempt": str(attempt),
            "time": str(time.time()),
        }
        await self.client.xadd(dlq, payload)
        await self.client.xack(stream, group, event_id)

    async def close(self):
        try:
            await self.client.close()
        except Exception:
            pass
```

5) app/gateways/retry_gateway.py
```python
import time

MAX_RETRIES = 3
RETRY_TTL = 86400  # seconds


class RetryGateway:
    """
    Simple retry counter using Redis INCR + EXPIRE.
    The provided redis client must be a redis.asyncio.Redis instance.
    """

    def __init__(self, redis_client):
        self.client = redis_client

    def _key(self, stream: str, event_id: str) -> str:
        return f"retry:{stream}:{event_id}"

    async def get_attempt(self, stream: str, event_id: str) -> int:
        key = self._key(stream, event_id)
        value = await self.client.incr(key)
        await self.client.expire(key, RETRY_TTL)
        return int(value)

    async def can_retry(self, stream: str, event_id: str) -> (bool, int):
        attempt = await self.get_attempt(stream, event_id)
        return (attempt <= MAX_RETRIES, attempt)
```

```python
async def run_node(self, ctx: Context, node: str):
    # skip if already DONE (idempotency)
    node_state = await self.checkpoint.get_node(ctx.session_id, node)
    if node_state and node_state.get("status") == "DONE":
        # optionally load result into ctx
        ctx.results[node] = node_state.get("result")
        return

    # emit START via event gateway (optional)
    await self.event_gateway.emit(f"events:{ctx.session_id}", {"node": node, "status": "START"})

    agent = self.registry.get(node)
    result = await agent(ctx) if asyncio.iscoroutinefunction(agent) else agent(ctx)

    # Save checkpoint atomically (hash + audit)
    await self.checkpoint.save_node_result(ctx.session_id, node, result, worker="runtime")

    # put result to ctx
    ctx.results[node] = result

    # emit DONE
    await self.event_gateway.emit(f"events:{ctx.session_id}", {"node": node, "status": "DONE", "result": result})

    # continue DAG
    for next_node in self.dag.next(node):
        await self.run_node(ctx, next_node)
```

Пропонований обсяг (що я реалізую в одному великому патчі)
1) Infrastructure layer
- app/infrastructure/redis_provider.py — єдиний connection pool (redis.asyncio), lifecycle (init/close)
- app/infrastructure/__init__.py

2) Gateway Layer (використовує RedisProvider)
- app/gateways/event_gateway.py (оbstracted EventGateway)
- app/gateways/checkpoint_gateway.py (HASH + audit XADD; MULTI/EXEC; optional Lua helper)
- app/gateways/retry_gateway.py (per-session/node retry counters + policy)
- app/gateways/dlq_gateway.py
- app/gateways/session_gateway.py (session meta: create/list/get/update)
- app/gateways/__init__.py (пакетний експорт)

3) Checkpoint / Recovery
- app/runtime/recovery.py — recover(session_id) logic that loads checkpoint and resumes DAG
- app/checkpoint_helpers.py — helper utilities (atomic save, get node, mark pending, set TTL)

4) Worker v2 (simplified, worker unaware of redis internals)
- workers/worker_v2.py — uses EventGateway + RetryGateway + DLQ + CheckpointGateway via Infrastructure; receives events, calls runtime.execute(event) and ack/handle retries/dlq.

5) Runtime integration
- app/runtime.py — adapt run_node to use gateways + idempotency checks + save_node_result before ack; small refactor to call recovery on start if checkpoint exists.

6) Test scaffolding / Fake gateways
- tests/unit/gateway/ (FakeRedis, FakeGateways)
- tests/integration/worker_recovery.py (uses FakeGateways)
- tests/unit/checkpoint_test.py (verify atomic writes and recovery logic)

7) Dev utilities
- scripts/seed_demo.py (create demo session)
- README update: how to run tests and docker compose notes

Ключові design decisions (рекомендую і буду використовувати за замовчуванням)
- session_id: UUIDv4 (але підтримка human alias: optional mapping session_alias:{alias} => uuid)
- Checkpoint: Redis HASH per session + state_audit:{sid} stream for immutable audit
- Atomic save: MULTI/EXEC pipeline; для найбільш критичних операцій додатково напишу Lua script (опціонально)
- Retry policy: retry:{sid}:{node} keys, default max_attempts=5, exponential backoff + jitter
- Parallel DAG: підтримка залежностей (node.depends list) — runtime/engine підтримує parallel ready set
- Worker: lightweight — лише consumes events and forwards to runtime.execute; all Redis IO through gateways

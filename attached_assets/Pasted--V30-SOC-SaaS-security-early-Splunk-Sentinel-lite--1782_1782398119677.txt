export function riskScore(event: string): number {
  const rules: Record<string, number> = {
    inject: 5,
    exploit: 6,
    scan: 3,
    bruteforce: 6,
    malware: 7,
    xss: 5,
    exfiltration: 8,
  };

  return Object.entries(rules).reduce((acc, [k, v]) => {
    return event.includes(k) ? acc + v : acc;
  }, 0);
}

export function decideAction(score: number) {
  if (score >= 12) return "ISOLATE";
  if (score >= 6) return "WARN";
  return "ALLOW";
}

export const wss = new WebSocketServer({ noServer: true });

export function broadcast(event: any) {
  wss.clients.forEach((client: any) => {
    if (client.readyState === 1) {
      client.send(JSON.stringify(event));
    }
  });
}

import { riskScore, decideAction } from "./engine";
import { broadcast } from "./eventBus";

export function processEvent(event: string, tenantId: number) {
  const score = riskScore(event);
  const action = decideAction(score);

  const payload = {
    tenantId,
    event,
    score,
    action,
    timestamp: Date.now(),
  };

  broadcast(payload);

  return payload;
}

🧠 4. RISK ENGINE (IMPROVED)
export function riskScore(event: string): number {
  const rules: Record<string, number> = {
    inject: 5,
    exploit: 6,
    scan: 3,
    bruteforce: 6,
    malware: 7,
    xss: 5,
    exfiltration: 8,
    lateral: 6,
    ransomware: 10,
  };

  let score = 0;

  for (const [k, v] of Object.entries(rules)) {
    if (event.toLowerCase().includes(k)) {
      score += v;
    }
  }

  return score;
}
TypeScript
export type Rule = {
  match: string;
  boost: number;
  forceAction?: "WARN" | "ISOLATE";
};

export const rules: Rule[] = [
  { match: "multiple login fail", boost: 4, forceAction: "WARN" },
  { match: "privilege escalation", boost: 8, forceAction: "ISOLATE" },
  { match: "data exfiltration", boost: 10, forceAction: "ISOLATE" },
];

export function applyRules(event: string) {
  let boost = 0;
  let forcedAction: any = null;

  for (const r of rules) {
    if (event.toLowerCase().includes(r.match)) {
      boost += r.boost;
      if (r.forceAction) forcedAction = r.forceAction;
    }
  }

  return { boost, forcedAction };
}

export function correlate(events: string[]) {
  const joined = events.join(" ").toLowerCase();

  if (joined.includes("scan") && joined.includes("exploit")) {
    return {
      threat: "CHAIN_ATTACK",
      severity: "HIGH",
      confidence: 0.9,
    };
  }

  if (joined.includes("bruteforce") && joined.includes("login")) {
    return {
      threat: "CREDENTIAL_STUFFING",
      severity: "MEDIUM",
      confidence: 0.75,
    };
  }

  return null;
}

export function autoFix(event: string): string | null {
  if (event.includes("inject")) return "INPUT_VALIDATION_PATCH";
  if (event.includes("bruteforce")) return "RATE_LIMITING_ENABLED";
  if (event.includes("xss")) return "OUTPUT_ENCODING_ENABLED";
  if (event.includes("exfiltration")) return "EGRESS_BLOCKED";
  if (event.includes("malware")) return "ISOLATION_MODE_ON";

  return null;
}
import crypto from "crypto";
import { ProcessedEvent } from "../types/soc";

export class MemoryChain {
  private chain: any[] = [];

  add(data: ProcessedEvent) {
    const prev = this.chain.length
      ? this.chain[this.chain.length - 1].hash
      : "GENESIS";

    const block = {
      ...data,
      prev,
      timestamp: Date.now(),
    };

    const hash = crypto
      .createHash("sha256")
      .update(JSON.stringify(block))
      .digest("hex");

    const finalBlock = { ...block, hash };

    this.chain.push(finalBlock);
    return finalBlock;
  }

  getAll() {
    return this.chain;
  }
}

export const memory = new MemoryChain();

import { riskScore } from "./riskEngine";
import { applyRules } from "./rulesEngine";
import { decideAction } from "./decisionEngine";
import { correlate } from "./correlationEngine";
import { autoFix } from "./autoFix";
import { memory } from "./memoryChain";

export function processPipeline(tenantId: number, event: string) {
  const baseScore = riskScore(event);
  const rule = applyRules(event);

  const finalScore = baseScore + rule.boost;

  const correlation = correlate([event]);

  const action =
    rule.forcedAction || decideAction(finalScore);

  const fix = action === "ALLOW" ? autoFix(event) : null;

  const stored = memory.add({
    tenantId,
    event,
    score: finalScore,
    action,
    timestamp: Date.now(),
  });

  return {
    tenantId,
    event,
    score: finalScore,
    action,
    correlation,
    fix,
    stored,
  };
}

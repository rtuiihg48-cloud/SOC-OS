```python name=app/storage/postgres.py
import os
import logging
from typing import AsyncGenerator

from sqlalchemy.ext.asyncio import create_async_engine, AsyncSession, async_sessionmaker

log = logging.getLogger("meta-cube.postgres")
log.setLevel(logging.INFO)


class PostgresProvider:
    """
    PostgresProvider supplies async engine and session factory.
    Use as a long-lived provider (one per process). Example usage:

    provider = PostgresProvider()
    await provider.init()
    async with provider.session() as session:
        ... use AsyncSession ...
    await provider.close()
    """

    def __init__(self, url: str | None = None):
        self.url = url or os.getenv(
            "DATABASE_URL",
            "postgresql+asyncpg://postgres:postgres@postgres:5432/metacube",
        )
        # tuning: tune pool_size / max_overflow via kwargs as needed
        self._engine = create_async_engine(self.url, pool_pre_ping=True)
        self._session_factory = async_sessionmaker(self._engine, expire_on_commit=False, class_=AsyncSession)
        self._inited = False

    async def init(self):
        # Optionally test connectivity
        if self._inited:
            return
        try:
            async with self._engine.begin() as conn:
                await conn.run_sync(lambda sync_conn: None)
            log.info("PostgresProvider connected to %s", self.url)
            self._inited = True
        except Exception as e:
            log.exception("Failed to init PostgresProvider: %s", e)
            raise

    def session(self) -> AsyncGenerator[AsyncSession, None]:
        """
        Async context manager to acquire an AsyncSession.

        Usage:
            async with provider.session() as session:
                await session.execute(...)
        """
        return self._session_factory()

    async def close(self):
        await self._engine.dispose()
        self._inited = False
        log.info("PostgresProvider disposed")
```

from datetime import datetime
from typing import Optional, List

from sqlalchemy.future import select
from sqlalchemy import update

from .models import SessionModel, ExecutionModel, AuditEventModel
from sqlalchemy.ext.asyncio import AsyncSession


class SessionRepository:
    def __init__(self, db: AsyncSession):
        self.db = db

    async def create(self, tenant_id: str, workflow: str, metadata: dict | None = None) -> SessionModel:
        session = SessionModel(tenant_id=tenant_id, workflow=workflow, metadata_=(metadata or {}))
        self.db.add(session)
        await self.db.commit()
        await self.db.refresh(session)
        return session

    async def get(self, session_id: str) -> Optional[SessionModel]:
        result = await self.db.execute(select(SessionModel).where(SessionModel.id == session_id))
        return result.scalar_one_or_none()

    async def update_status(self, session_id: str, status: str) -> None:
        stmt = update(SessionModel).where(SessionModel.id == session_id).values(status=status, updated_at=datetime.utcnow())
        await self.db.execute(stmt)
        await self.db.commit()


class ExecutionRepository:
    def __init__(self, db: AsyncSession):
        self.db = db

    async def save(self, session_id: str, node: str, status: str, result: dict | None = None, error: str | None = None) -> ExecutionModel:
        execution = ExecutionModel(session_id=session_id, node=node, status=status, result=(result or {}), error=error)
        self.db.add(execution)
        await self.db.commit()
        await self.db.refresh(execution)
        return execution

    async def list_session(self, session_id: str) -> List[ExecutionModel]:
        result = await self.db.execute(select(ExecutionModel).where(ExecutionModel.session_id == session_id))
        return result.scalars().all()


class AuditRepository:
    def __init__(self, db: AsyncSession):
        self.db = db

    async def append(self, session_id: str, event: str, payload: dict) -> AuditEventModel:
        item = AuditEventModel(session_id=session_id, event=event, payload=payload)
        self.db.add(item)
        await self.db.commit()
        await self.db.refresh(item)
        return item
